open suspend fun listSessions(
    namUser: String,
    page: Int,
    size: Int,
    status: SessionStatus?,
    appName: String?,
): PageResponse<SessionSummaryResponse> {

    logger.info(
        "Listing user sessions: namUser={}, page={}, size={}, status={}, appName={}",
        namUser,
        page,
        size,
        status,
        appName,
    )

    /*
     * Neste momento, iara:sessions:{userId} é paginado antes
     * de acessarmos a metadata.
     *
     * Portanto não podemos aplicar filtros de status/system
     * depois da paginação sem comprometer total/páginas.
     *
     * Para consultas filtradas mantemos Oracle como fallback.
     */
    if (
        status != null ||
        !appName.isNullOrBlank()
    ) {
        return listSessionsFromOracle(
            namUser = namUser,
            page = page,
            size = size,
            status = status,
            appName = appName,
        )
    }

    /*
     * Redis-first:
     *
     * iara:sessions:{userId}
     */
    val sessionIdsPage =
        memoryService.listSessionIds(
            userId = namUser,
            page = page,
            size = size,
        )

    /*
     * Índice ausente/vazio:
     * usamos a fonte durável.
     */
    if (sessionIdsPage.content.isEmpty()) {
        return listSessionsFromOracle(
            namUser = namUser,
            page = page,
            size = size,
            status = null,
            appName = null,
        )
    }

    /*
     * Para cada sessionId:
     *
     * iara:session-metadata:{sessionId}
     */
    val metadataBySessionId =
        sessionIdsPage.content
            .mapNotNull { sessionId ->

                memoryService
                    .findSessionMetadata(sessionId)
                    ?.let { metadata ->
                        sessionId to metadata
                    }
            }
            .toMap()

    /*
     * Se alguma metadata não estiver no Redis,
     * não inventamos os dados.
     *
     * Como pode ter ocorrido TTL/flush/falha parcial,
     * fazemos fallback para Oracle.
     */
    if (metadataBySessionId.size != sessionIdsPage.content.size) {

        logger.warn(
            "Session metadata incomplete in Redis. Falling back to Oracle: namUser={}",
            namUser,
        )

        return listSessionsFromOracle(
            namUser = namUser,
            page = page,
            size = size,
            status = null,
            appName = null,
        )
    }

    /*
     * IMPORTANTE:
     *
     * Mantemos a ordem fornecida por
     * iara:sessions:{userId}.
     *
     * Não usamos metadataBySessionId.values,
     * pois isso poderia perder a ordem do ZSET.
     */
    val content =
        sessionIdsPage.content.map { sessionId ->

            val metadata =
                requireNotNull(
                    metadataBySessionId[sessionId]
                )

            metadata.toSessionSummaryResponse()
        }

    return PageResponse(
        content = content,
        pageable = sessionIdsPage.pageable,
        totalElements = sessionIdsPage.totalElements,
        totalPages = sessionIdsPage.totalPages,
        first = sessionIdsPage.first,
        last = sessionIdsPage.last,
        empty = content.isEmpty(),
    )
}
