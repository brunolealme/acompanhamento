@Repository
open class RedisSessionMetadataRepository(
    private val redisTemplate: ReactiveStringRedisTemplate,
    private val serializer: MemoryJsonSerializer,
    private val properties: IaraMemoryProperties,
) : SessionMetadataRepository {

    override suspend fun save(
        metadata: SessionMetadata,
    ) {
        val key =
            CacheConstants.sessionMetadataKey(
                keyPrefix = properties.keyPrefix,
                sessionId = metadata.sessionId,
            )

        val value =
            serializer.serializeSessionMetadata(metadata)

        redisTemplate
            .opsForValue()
            .set(
                key,
                value,
                properties.sessionTtl,
            )
            .awaitSingle()
    }

    override suspend fun findBySessionId(
        sessionId: String,
    ): SessionMetadata? {
        val key =
            CacheConstants.sessionMetadataKey(
                keyPrefix = properties.keyPrefix,
                sessionId = sessionId,
            )

        val payload =
            redisTemplate
                .opsForValue()
                .get(key)
                .awaitSingleOrNull()
                ?: return null

        return serializer.deserializeSessionMetadata(payload)
    }

    override suspend fun delete(
        sessionId: String,
    ) {
        val key =
            CacheConstants.sessionMetadataKey(
                keyPrefix = properties.keyPrefix,
                sessionId = sessionId,
            )

        redisTemplate
            .delete(key)
            .awaitSingle()
    }
}

import kotlinx.coroutines.reactive.awaitSingleOrNull
import org.springframework.data.redis.core.ReactiveStringRedisTemplate

@Component
class MemoryJsonSerializer(
    private val objectMapper: ObjectMapper,
) {

    fun serializeConversationTurn(
        turn: ConversationTurn,
    ): String =
        objectMapper.writeValueAsString(turn)

    fun deserializeConversationTurn(
        payload: String,
    ): ConversationTurn =
        objectMapper.readValue(payload)

    fun serializeSessionMetadata(
        metadata: SessionMetadata,
    ): String =
        objectMapper.writeValueAsString(metadata)

    fun deserializeSessionMetadata(
        payload: String,
    ): SessionMetadata =
        objectMapper.readValue(payload)
}

open class RedisSessionMetadataRepository(

    private val serializer: MemoryJsonSerializer,

                properties.keyPrefix,
                metadata.sessionId,

            .opsForValue()

    override suspend fun findBySessionId(
        sessionId: String,

            CacheConstants.sessionMetadataKey(
