@Service
open class MemoryService(
    private val sessionMemoryRepository: SessionMemoryRepository,
    private val sessionIndexRepository: SessionIndexRepository,
    private val contextCacheRepository: ContextCacheRepository,
    private val failOpen: FailOpen,
) {

    open suspend fun appendTurn(
        turn: ConversationTurn,
        userId: String?,
    ) {
        /*
         * Obrigatórios:
         *
         * iara:memory:{sessionId}
         * iara:order:{sessionId}
         */
        sessionMemoryRepository.appendTurn(
            sessionId = turn.sessionId,
            turn = turn,
        )

        /*
         * Obrigatório para posterior persistência no Oracle.
         *
         * iara:pending
         * chatHistoryId -> sessionId
         */
        sessionMemoryRepository.addPendingTurn(
            sessionId = turn.sessionId,
            chatHistoryId = turn.chatHistoryId,
        )

        /*
         * Índice auxiliar para listagem das sessões do usuário.
         *
         * iara:sessions:{userId}
         */
        userId?.let {
            failOpen.failOpenWrite(
                operation = "touchUserSession",
            ) {
                sessionIndexRepository.touch(
                    userId = it,
                    sessionId = turn.sessionId,
                )
            }
        }

        /*
         * Um novo turno torna o contexto cacheado obsoleto.
         */
        failOpen.failOpenWrite(
            operation = "evictConversationContext",
        ) {
            contextCacheRepository.evict(
                turn.sessionId,
            )
        }
    }

    open suspend fun getConversationContext(
        sessionId: String,
        limit: Int? = null,
    ): String? {
        val cachedContext =
            failOpen.failOpenRead(
                operation = "findConversationContext",
                default = null,
            ) {
                contextCacheRepository.find(sessionId)
            }

        if (!cachedContext.isNullOrBlank()) {
            return cachedContext
        }

        val turns =
            failOpen.failOpenRead(
                operation = "findRecentTurns",
                default = emptyList(),
            ) {
                sessionMemoryRepository.findRecentTurns(
                    sessionId = sessionId,
                    limit = limit,
                )
            }

        if (turns.isEmpty()) {
            return null
        }

        val context =
            buildConversationContext(turns)

        failOpen.failOpenWrite(
            operation = "saveConversationContext",
        ) {
            contextCacheRepository.save(
                sessionId = sessionId,
                context = context,
            )
        }

        return context
    }

    open suspend fun listTurns(
        sessionId: String,
        page: Int,
        size: Int,
    ): PageResponse<ConversationTurn> =
        failOpen.failOpenRead(
            operation = "listTurns",
            default = PageResponse.of(
                content = emptyList(),
                total = 0,
                page = page,
                size = size,
            ),
        ) {
            sessionMemoryRepository.listTurns(
                sessionId = sessionId,
                page = page,
                size = size,
            )
        }

    open suspend fun findTurn(
        sessionId: String,
        chatHistoryId: String,
    ): ConversationTurn? =
        failOpen.failOpenRead(
            operation = "findTurn",
            default = null,
        ) {
            sessionMemoryRepository.findTurn(
                sessionId = sessionId,
                chatHistoryId = chatHistoryId,
            )
        }

    open suspend fun listSessionIds(
        userId: String,
        page: Int,
        size: Int,
    ): PageResponse<String> =
        failOpen.failOpenRead(
            operation = "listSessionIds",
            default = PageResponse.of(
                content = emptyList(),
                total = 0,
                page = page,
                size = size,
            ),
        ) {
            sessionIndexRepository.listByUser(
                userId = userId,
                page = page,
                size = size,
            )
        }

    open suspend fun findPendingTurns(
        limit: Long,
    ): List<PendingTurnReference> =
        sessionMemoryRepository.findPendingTurns(
            limit = limit,
        )

    open suspend fun markTurnAsPersisted(
        sessionId: String,
        chatHistoryId: String,
    ): Boolean =
        sessionMemoryRepository.updateTurnPersistenceStatus(
            sessionId = sessionId,
            chatHistoryId = chatHistoryId,
            persistStatus = PersistStatus.PERSISTED,
        )

    open suspend fun evictConversationContext(
        sessionId: String,
    ) {
        failOpen.failOpenWrite(
            operation = "evictConversationContext",
        ) {
            contextCacheRepository.evict(sessionId)
        }
    }

    private fun buildConversationContext(
        turns: List<ConversationTurn>,
    ): String =
        turns.joinToString(separator = "\n\n") { turn ->
            buildString {
                append("Usuário: ")
                append(turn.question)

                turn.answer
                    ?.takeIf { it.isNotBlank() }
                    ?.let { answer ->
                        append("\nAssistente: ")
                        append(answer)
                    }
            }
        }
}
