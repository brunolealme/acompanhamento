@Repository
open class RedisContextCacheRepository(
    private val redisTemplate: ReactiveStringRedisTemplate,
    private val properties: IaraMemoryProperties,
) : ContextCacheRepository {

    override suspend fun save(
        sessionId: String,
        context: String,
    ) {
        validateSessionId(sessionId)

        val key =
            CacheConstants.contextKey(
                properties.keyPrefix,
                sessionId,
            )

        redisTemplate
            .opsForHash<String, String>()
            .put(
                key,
                CONTEXT_FIELD,
                context,
            )
            .awaitSingle()

        redisTemplate
            .expire(
                key,
                properties.contextCache.ttl,
            )
            .awaitSingle()
    }

    override suspend fun find(
        sessionId: String,
    ): String? {
        validateSessionId(sessionId)

        val key =
            CacheConstants.contextKey(
                properties.keyPrefix,
                sessionId,
            )

        return redisTemplate
            .opsForHash<String, String>()
            .get(
                key,
                CONTEXT_FIELD,
            )
            .awaitFirstOrNull()
    }

    override suspend fun evict(
        sessionId: String,
    ) {
        validateSessionId(sessionId)

        val key =
            CacheConstants.contextKey(
                properties.keyPrefix,
                sessionId,
            )

        redisTemplate
            .delete(key)
            .awaitSingle()
    }

    private fun validateSessionId(
        sessionId: String,
    ) {
        require(
            sessionId.length <= MAX_IDENTIFIER_LENGTH
        ) {
            "sessionId must have at most $MAX_IDENTIFIER_LENGTH characters"
        }
    }

    companion object {
        private const val CONTEXT_FIELD = "context"
        private const val MAX_IDENTIFIER_LENGTH = 40
    }
}

import kotlinx.coroutines.reactor.awaitSingle
import kotlinx.coroutines.reactor.awaitFirstOrNull
import org.springframework.data.redis.core.ReactiveStringRedisTemplate
import org.springframework.stereotype.Repository


-----
interface ContextCacheRepository {

    suspend fun save(
        sessionId: String,
        context: String,
    )

    suspend fun find(
        sessionId: String,
    ): String?

    suspend fun evict(
        sessionId: String,
    )
}
