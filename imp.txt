interface SessionIndexRepository {

    suspend fun touch(
        userId: String,
        sessionId: String,
        at: Instant = Instant.now(),
    )

    suspend fun listByUser(
        userId: String,
        page: Int,
        size: Int,
    ): PageResponse<String>
}

----

@Repository
open class RedisSessionIndexRepository(
    private val redisTemplate: ReactiveStringRedisTemplate,
    private val properties: IaraMemoryProperties,
) : SessionIndexRepository {

    override suspend fun touch(
        userId: String,
        sessionId: String,
        at: Instant,
    ) {
        validateIdentifiers(
            userId = userId,
            sessionId = sessionId,
        )

        val key =
            CacheConstants.userSessionsKey(
                properties.keyPrefix,
                userId,
            )

        redisTemplate
            .opsForZSet()
            .add(
                key,
                sessionId,
                at.toEpochMilli().toDouble(),
            )
            .awaitSingle()

        redisTemplate
            .expire(
                key,
                properties.sessionTtl,
            )
            .awaitSingle()
    }

    override suspend fun listByUser(
        userId: String,
        page: Int,
        size: Int,
    ): PageResponse<String> {
        require(userId.length <= MAX_IDENTIFIER_LENGTH) {
            "userId must have at most $MAX_IDENTIFIER_LENGTH characters"
        }

        val normalizedPage =
            PageUtil.normalizePage(page)

        val normalizedSize =
            PageUtil.normalizeSizeCoerceIn(
                size,
                properties.maxPageSize,
            )

        val key =
            CacheConstants.userSessionsKey(
                properties.keyPrefix,
                userId,
            )

        val total =
            redisTemplate
                .opsForZSet()
                .size(key)
                .awaitSingle()

        if (total == 0L) {
            return PageResponse.of(
                content = emptyList(),
                total = 0,
                page = normalizedPage,
                size = normalizedSize,
            )
        }

        val start =
            normalizedPage.toLong() *
                normalizedSize.toLong()

        val end =
            start + normalizedSize.toLong() - 1

        val sessionIds =
            redisTemplate
                .opsForZSet()
                .reverseRange(
                    key,
                    Range.closed(start, end),
                )
                .collectList()
                .awaitSingle()

        return PageResponse.of(
            content = sessionIds,
            total = total,
            page = normalizedPage,
            size = normalizedSize,
        )
    }

    private fun validateIdentifiers(
        userId: String,
        sessionId: String,
    ) {
        require(userId.length <= MAX_IDENTIFIER_LENGTH) {
            "userId must have at most $MAX_IDENTIFIER_LENGTH characters"
        }

        require(sessionId.length <= MAX_IDENTIFIER_LENGTH) {
            "sessionId must have at most $MAX_IDENTIFIER_LENGTH characters"
        }
    }

    companion object {
        private const val MAX_IDENTIFIER_LENGTH = 40
    }
}
