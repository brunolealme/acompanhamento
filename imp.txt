@Component
@ConditionalOnBean(MemoryService::class)
open class RedisBackedConversationWindowSource(
    private val memoryService: MemoryService,
) : ConversationWindowSource {

    override suspend fun recentWindow(
        sessionId: String,
        maxMessages: Int,
    ): List<MemoryTurn> {
        require(maxMessages >= 0) {
            "maxMessages must be greater than or equal to zero"
        }

        if (maxMessages == 0) {
            return emptyList()
        }

        return memoryService
            .findRecentTurns(
                sessionId = sessionId,
                limit = maxMessages,
            )
            .map { turn ->
                MemoryTurn(
                    question = turn.question,
                    answer = turn.answer,
                    agentName = turn.agentName,
                    category = turn.category,
                    createdAt = turn.createdAt,
                )
            }
    }
}
