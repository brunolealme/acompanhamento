@Service
open class MemoryService(
    private val sessionMemoryRepository: SessionMemoryRepository,
    private val sessionIndexRepository: SessionIndexRepository,
    private val contextCacheRepository: ContextCacheRepository,
    private val failOpen: FailOpen,
) {

    open suspend fun getConversationContext(
        sessionId: String,
        window: Int = 20,
    ): String? {
        val cachedContext =
            failOpen.failOpenRead(
                operation = "getConversationContextCache",
                default = null,
            ) {
                contextCacheRepository.get(sessionId)
            }

        if (!cachedContext.isNullOrBlank()) {
            return cachedContext
        }

        val turns =
            failOpen.failOpenRead(
                operation = "getRecentTurns",
                default = emptyList(),
            ) {
                sessionMemoryRepository.getRecentTurns(
                    sessionId = sessionId,
                    window = window,
                )
            }

        if (turns.isEmpty()) {
            return null
        }

        val context =
            turns.joinToString("\n\n") { turn ->
                buildString {
                    appendLine("Usuário: ${turn.question}")

                    turn.answer
                        ?.takeIf { it.isNotBlank() }
                        ?.let {
                            append("Assistente: $it")
                        }
                }
            }

        failOpen.failOpenWrite(
            operation = "cacheConversationContext",
        ) {
            contextCacheRepository.put(
                sessionId = sessionId,
                value = context,
            )
        }

        return context
    }

    open suspend fun appendTurn(
        turn: ConversationTurn,
        metadata: SessionMetadata,
    ) {
        sessionMemoryRepository.appendTurn(
            sessionId = metadata.sessionId,
            turn = turn,
        )

        failOpen.failOpenWrite(
            operation = "markTurnPendingPersistence",
        ) {
            sessionMemoryRepository.insertPendingTurn(
                sessionId = metadata.sessionId,
                chatHistoryId = turn.chatHistoryId,
            )
        }

        failOpen.failOpenWrite(
            operation = "updateSessionIndex",
        ) {
            sessionIndexRepository.upsertSessionOnTurn(metadata)
        }

        failOpen.failOpenWrite(
            operation = "invalidateConversationContext",
        ) {
            contextCacheRepository.evict(metadata.sessionId)
        }
    }
}
