@Service
class ChatService(
    private val memoryService: MemoryService,
) {

    fun processQuestion(
        request: ChatAskRequest,
        appName: String?,
        correlationId: String?,
    ): Flow<ChatStreamEvent> = channelFlow {

        val sessionId =
            request.sessionId
                ?: UUID.randomUUID().toString()

        val chatHistoryId =
            UUID.randomUUID().toString()

        val conversationContext =
            memoryService.getConversationContext(
                sessionId = sessionId,
            )

        send(
            ChatStreamEvent(
                type = "session",
                sessionId = sessionId,
            )
        )

        val completeResponse =
            generateMockResponse(
                question = request.question,
                requestContext = request.context,
                conversationContext = conversationContext,
            )

        val answerBuilder = StringBuilder()

        completeResponse
            .chunked(CHUNK_SIZE)
            .forEach { chunk ->

                delay(CHUNK_DELAY)

                answerBuilder.append(chunk)

                send(
                    ChatStreamEvent(
                        type = "chunk",
                        sessionId = sessionId,
                        content = chunk,
                    )
                )
            }

        val turn =
            ConversationTurn(
                sessionId = sessionId,
                chatHistoryId = chatHistoryId,
                question = request.question,
                answer = answerBuilder.toString(),
                persistStatus = PersistStatus.PENDING,
                // demais campos reais do seu model
            )

        memoryService.appendTurn(
            turn = turn,
            userId = request.namUser,
        )

        send(
            ChatStreamEvent(
                type = "done",
                sessionId = sessionId,
            )
        )
    }

    private fun generateMockResponse(
        question: String,
        requestContext: Map<String, String>?,
        conversationContext: String?,
    ): String {
        // ÚNICO MOCK.
        // Mantemos aqui enquanto não houver integração real com o agente.

        return """
            Pergunta recebida: $question

            Contexto da requisição:
            ${requestContext.orEmpty()}

            Contexto da conversa:
            ${conversationContext ?: "Sem histórico anterior."}

            Esta é uma resposta simulada do agente Iara.
        """.trimIndent()
    }

    companion object {
        private const val CHUNK_SIZE = 20
        private const val CHUNK_DELAY = 200L
    }
}
