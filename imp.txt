@Service
open class MemoryService(
    private val sessionMemoryRepository: SessionMemoryRepository,
    private val sessionIndexRepository: SessionIndexRepository,
    private val contextCacheRepository: ContextCacheRepository,
    private val failOpen: FailOpen,
) {

    companion object {
        private const val DEFAULT_CONTEXT_WINDOW = 20
    }

    open suspend fun appendTurn(
        turn: ConversationTurn,
        metadata: SessionMetadata,
    ) {
        sessionMemoryRepository.appendTurn(
            sessionId = metadata.sessionId,
            turn = turn,
        )

        failOpen.failOpenWrite(
            operation = "updateSessionIndex",
        ) {
            sessionIndexRepository.upsertSessionOnTurn(metadata)
        }

        failOpen.failOpenWrite(
            operation = "invalidateContextCache",
        ) {
            contextCacheRepository.evict(metadata.sessionId)
        }
    }

    open suspend fun getConversationContext(
        sessionId: String,
    ): String? {
        val cachedContext =
            failOpen.failOpenRead(
                operation = "getConversationContextCache",
                default = null,
            ) {
                contextCacheRepository.get(sessionId)
            }

        if (!cachedContext.isNullOrBlank()) {
            return cachedContext
        }

        val turns =
            getRecentTurns(
                sessionId = sessionId,
                limit = DEFAULT_CONTEXT_WINDOW,
            )

        if (turns.isEmpty()) {
            return null
        }

        val context = buildConversationContext(turns)

        failOpen.failOpenWrite(
            operation = "cacheConversationContext",
        ) {
            contextCacheRepository.put(
                sessionId = sessionId,
                value = context,
            )
        }

        return context
    }

    open suspend fun getRecentTurns(
        sessionId: String,
        limit: Int = DEFAULT_CONTEXT_WINDOW,
    ): List<ConversationTurn> {
        return failOpen.failOpenRead(
            operation = "getRecentTurns",
            default = emptyList(),
        ) {
            sessionMemoryRepository.getRecentTurns(
                sessionId = sessionId,
                window = limit,
            )
        }
    }

    open suspend fun listTurns(
        sessionId: String,
        page: Int,
        size: Int,
    ): PageResponse<ConversationTurn> {
        return failOpen.failOpenRead(
            operation = "listTurns",
            default = emptyTurnPage(
                page = page,
                size = size,
            ),
        ) {
            sessionMemoryRepository.getTurns(
                sessionId = sessionId,
                page = page,
                size = size,
            )
        }
    }

    open suspend fun listSessions(
        userId: String,
        page: Int,
        size: Int,
    ): PageResponse<SessionMetadata> {
        return failOpen.failOpenRead(
            operation = "listSessions",
            default = emptySessionPage(
                page = page,
                size = size,
            ),
        ) {
            sessionIndexRepository.listByUser(
                namUser = userId,
                page = page,
                size = size,
            )
        }
    }

    open suspend fun findSession(
        sessionId: String,
    ): SessionMetadata? {
        return failOpen.failOpenRead(
            operation = "findSession",
            default = null,
        ) {
            sessionIndexRepository.getSession(sessionId)
        }
    }

    open suspend fun findTurn(
        sessionId: String,
        chatHistoryId: String,
    ): ConversationTurn? {
        return failOpen.failOpenRead(
            operation = "findTurn",
            default = null,
        ) {
            sessionMemoryRepository.findTurn(
                sessionId = sessionId,
                chatHistoryId = chatHistoryId,
            )
        }
    }

    open suspend fun updateSessionStatus(
        sessionId: String,
        status: SessionStatus,
    ): Boolean {
        return failOpen.failOpenWrite(
            operation = "updateSessionStatus",
            default = false,
        ) {
            sessionIndexRepository.updateStatus(
                sessionId = sessionId,
                status = status,
            )
        }
    }

    open suspend fun resetConversationContext(
        sessionId: String,
        metadata: SessionMetadata,
    ) {
        failOpen.failOpenWrite(
            operation = "clearConversationTurns",
        ) {
            sessionMemoryRepository.clearTurns(sessionId)
        }

        failOpen.failOpenWrite(
            operation = "updateSessionIndexAfterContextReset",
        ) {
            sessionIndexRepository.upsertSessionOnTurn(metadata)
        }

        failOpen.failOpenWrite(
            operation = "evictConversationContextCache",
        ) {
            contextCacheRepository.evict(sessionId)
        }
    }

    private fun buildConversationContext(
        turns: List<ConversationTurn>,
    ): String {
        return turns.joinToString("\n\n") { turn ->
            buildString {
                appendLine("Usuário: ${turn.question}")

                turn.answer
                    ?.takeIf { it.isNotBlank() }
                    ?.let {
                        append("Assistente: $it")
                    }
            }
        }
    }

    private fun emptyTurnPage(
        page: Int,
        size: Int,
    ): PageResponse<ConversationTurn> {
        return PageResponse(
            content = emptyList(),
            total = 0,
            page = page,
            size = size,
        )
    }

    private fun emptySessionPage(
        page: Int,
        size: Int,
    ): PageResponse<SessionMetadata> {
        return PageResponse(
            content = emptyList(),
            total = 0,
            page = page,
            size = size,
        )
    }
}

fun processQuestion(
    request: ChatAskRequest,
    appName: String?,
    correlationId: String?,
): Flow<ChatStreamEvent> = channelFlow {

    val sessionId =
        request.sessionId ?: UUID.randomUUID().toString()

    val conversationContext =
        memoryService.getConversationContext(sessionId)

    val response =
        generateMockResponse(
            question = request.question,
            requestContext = request.context,
            conversationContext = conversationContext,
        )

    val answer = StringBuilder()

    response.chunked(20).forEach { chunk ->

        delay(200)

        answer.append(chunk)

        send(
            ChatStreamEvent.chunk(
                sessionId = sessionId,
                content = chunk,
            )
        )
    }

    val turn =
        buildTurn(
            sessionId = sessionId,
            request = request,
            answer = answer.toString(),
        )

    val session =
        buildSessionMetadata(
            sessionId = sessionId,
            request = request,
            appName = appName,
        )

    memoryService.appendTurn(
        turn = turn,
        session = session,
    )

    send(
        ChatStreamEvent.done(sessionId)
    )
}
