open suspend fun getSession(
    sessionId: String,
    page: Int,
    size: Int,
    appName: String?,
): ChatSessionResponse {
    logger.info(
        "Recovering session: sessionId={}, page={}, size={}, appName={}",
        sessionId,
        page,
        size,
        appName,
    )

    /*
     * HISTÓRICO
     *
     * Redis primeiro.
     */
    val redisHistory =
        memoryService.listTurns(
            sessionId = sessionId,
            page = page,
            size = size,
        )

    val messages =
        if (redisHistory.content.isNotEmpty()) {
            redisHistory.toHistoryResponse()
        } else {
            /*
             * Redis miss / TTL / indisponibilidade:
             * consulta fonte durável.
             */
            findHistoryInOracle(
                sessionId = sessionId,
                page = page,
                size = size,
            )
        }

    /*
     * METADATA
     *
     * Redis primeiro.
     */
    val metadata =
        sessionMetadataRepository.findById(sessionId)
            ?: sessionRepository
                .findById(sessionId)
                ?.let { session ->
                    SessionMetadata(
                        sessionId = session.idtSession,
                        status = session.flgStatus,
                        system = session.namSystem,
                        createdAt = session.datCreated,
                    )
                }
            ?: throw ResponseStatusException(
                HttpStatus.NOT_FOUND,
                "Session not found: $sessionId",
            )

    return ChatSessionResponse(
        sessionId = metadata.sessionId,
        status = metadata.status,
        system = metadata.system,
        createdAt = metadata.createdAt,
        messages = messages,
    )
}
